//Configuración inicial de Git en GitBash//

$ git --versión 
//Muestra la versión que tenemos instalada

$ git config --global user.name "Nuestro nombre"
$ git config --global user.email Nuestro email

//--global sirve para indicar que la configuración se realizará de forma global y no por proyecto//
//En el correo no es necesario las comillas//

$ git config --global core.editor "code --wait"
//Configurar VSCode como nuestro editor principal//
//La opción de --wait es para que la terminal se quede esperando hasta que cerremos el editor//

$ git config --global -e
//Muestra el archivo de configuración global en el editor//

$ git config --global core.autocrlf true/input
//Configura el salto de linea//
//Para linux o Mac se ocupa input
//Para Windows se ocupa true
